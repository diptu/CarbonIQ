from typing import Annotated, Tuple
from fastapi import Depends, Header, HTTPException, status
from jose import JWTError
from sqlalchemy.ext.asyncio import AsyncSession

from app.core import security
from app.core.exceptions import UnauthorizedException
from app.models.user import User
from .db import get_async_db


async def get_current_user_and_tenant(
    token: Annotated[str, Header(alias="Authorization")],
    db: Annotated[AsyncSession, Depends(get_async_db)],
) -> Tuple[User, str]:
    """Validate JWT, fetch User, and extract Tenant ID."""
    if not token or not token.startswith("Bearer "):
        raise UnauthorizedException("Missing or malformed Authorization header.")

    token_value = token.split(" ", 1)[1]
    payload = security.verify_token(token_value)
    if not payload:
        raise UnauthorizedException("Token is invalid or expired.")

    user_id, tenant_id, token_type = (
        payload.get("user_id"),
        payload.get("tenant_id"),
        payload.get("type"),
    )
    if token_type != "access" or not all([user_id, tenant_id]):
        raise UnauthorizedException("Invalid token claims or not an access token.")

    user = await db.get(User, user_id)
    if not user or not user.is_active:
        raise UnauthorizedException("User not found or inactive.")
    return user, tenant_id


async def get_current_user(
    data: Annotated[Tuple[User, str], Depends(get_current_user_and_tenant)],
) -> User:
    """Return the authenticated User."""
    return data[0]


async def get_current_tenant_id(
    data: Annotated[Tuple[User, str], Depends(get_current_user_and_tenant)],
) -> str:
    """Return the authenticated Tenant ID."""
    return data[1]

from typing import AsyncGenerator
from app.db.session import get_db
from sqlalchemy.ext.asyncio import AsyncSession


async def get_async_db() -> AsyncGenerator[AsyncSession, None]:
    """
    Dependency wrapper around core get_db() for FastAPI routes.

    Handles commit/rollback automatically.
    """
    async with get_db() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()
"""
Tenant-scoped, DB-driven RBAC for FastAPI.

Features:
- Per-request permission caching (ContextVar)
- Policy hooks support
- Audit logging for all permission checks
"""

from typing import Annotated, List, Set, Dict
from fastapi import Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from contextvars import ContextVar

from ..models.user import User
from ..services.rbac_service import RBACService, PermissionError
from ..services.role_service import RoleService
from ..services.permission_service import PermissionService
from ..services.audit_adapter import AuditAdapter
from .auth import get_current_user, get_current_tenant_id
from .db import get_async_db

# ----------------------------
# Context cache for per-request user permissions
# ----------------------------
_user_permission_cache: ContextVar[Dict[str, Set[str]]] = ContextVar(
    "_user_permission_cache", default={}
)

# ----------------------------
# Audit Adapter instance
# ----------------------------
audit = AuditAdapter()


# ----------------------------
# RBAC Service Provider
# ----------------------------
def get_rbac_service(db: Annotated[AsyncSession, Depends(get_async_db)]) -> RBACService:
    role_service = RoleService(db=db)
    permission_service = PermissionService(db=db)
    return RBACService(
        role_service=role_service,
        permission_crud=permission_service,
    )


# ----------------------------
# Helper: fetch user permissions from DB
# ----------------------------
async def _fetch_user_permissions(
    user_id: str, tenant_id: str, rbac_service: RBACService
) -> Set[str]:
    user: User | None = await rbac_service.db.get(User, user_id)
    if not user:
        return set()

    perms: Set[str] = set()
    for role in getattr(user, "roles", []):
        if getattr(role, "tenant_id", None) != tenant_id:
            continue
        role_perms = await rbac_service.role_service.get_effective_permissions(str(role.id))
        perms.update(role_perms)

    return perms


# ----------------------------
# Dependency: require a specific permission
# ----------------------------
def require_permission(permission_code: str):
    async def dependency(
        user: Annotated[User, Depends(get_current_user)],
        tenant_id: Annotated[str, Depends(get_current_tenant_id)],
        rbac_service: Annotated[RBACService, Depends(get_rbac_service)],
    ) -> User:
        user_key = f"{tenant_id}:{user.id}"
        cache = _user_permission_cache.get()
        if user_key not in cache:
            cache[user_key] = await _fetch_user_permissions(str(user.id), tenant_id, rbac_service)
            _user_permission_cache.set(cache)

        granted = permission_code in cache[user_key]
        await audit.log(
            actor_id=str(user.id),
            tenant_id=tenant_id,
            action="permission_check",
            resource=permission_code,
            status=200 if granted else 403,
        )

        if not granted:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Insufficient permissions. Required: {permission_code}",
            )

        return user

    return dependency


# ----------------------------
# Dependency: require at least one of multiple permissions
# ----------------------------
def require_any_permission(required_permissions: List[str]):
    async def dependency(
        user: Annotated[User, Depends(get_current_user)],
        tenant_id: Annotated[str, Depends(get_current_tenant_id)],
        rbac_service: Annotated[RBACService, Depends(get_rbac_service)],
    ) -> User:
        user_key = f"{tenant_id}:{user.id}"
        cache = _user_permission_cache.get()
        if user_key not in cache:
            cache[user_key] = await _fetch_user_permissions(str(user.id), tenant_id, rbac_service)
            _user_permission_cache.set(cache)

        granted = any(p in cache[user_key] for p in required_permissions)
        await audit.log(
            actor_id=str(user.id),
            tenant_id=tenant_id,
            action="permission_check",
            resource=",".join(required_permissions),
            status=200 if granted else 403,
        )

        if not granted:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Insufficient permissions. Required one of: {required_permissions}",
            )

        return user

    return dependency
